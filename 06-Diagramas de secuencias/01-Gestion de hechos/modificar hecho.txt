@startuml
actor Administrador

participant "ch: ControladorHecho" as CH
participant "cdh: <Singleton> CatalogoDeHechos" as CatH
participant "hecho[i]: Hecho" as H
participant "hecho: Hecho" as He

== Modificar hecho ==

Administrador -> CH : modificar_hecho(id, nuevosDatos)
activate CH

CH -> CatH : buscar_hecho_por_id(id)
activate CatH

CatH -> He : getDatos()
activate He
He --> CatH : <<datos actuales>>
deactivate He

CatH --> CH : <<hecho encontrado>>
deactivate CatH

CH -> CH : validarDatos(nuevosDatos)

alt Datos inválidos
    CH --> Administrador : <<error de validación>>

else Datos válidos

    CH -> CatH : verificarNombre(nuevosDatos.nombre)
    activate CatH

    loop para cada hecho en CatalogoDeHechos

        CatH -> H : getNombre()
        activate H
        H --> CatH : <<nombre>>
            note right of CatH
                Si nombre = nombre_modificado
                hecho = hecho[i]
                cortar iteración
            end note
        deactivate H

    end

    CatH --> CH : <<hecho>>
    deactivate CatH

    alt Nombre disponible
        CH -> He : actualizar(nuevosDatos)
        activate He
        He --> CH : <<confirmación actualización>>
        deactivate He

        CH --> Administrador : <<hecho modificado>>

    else Nombre repetido / error
        CH --> Administrador : <<error: nombre existente>>
    end

end

deactivate CH
@enduml
