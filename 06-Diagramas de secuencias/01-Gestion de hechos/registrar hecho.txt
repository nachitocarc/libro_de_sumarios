@startuml
actor Administrador

participant "ch: ControladorHecho" as CH
participant "cdh: <Singleton> CatalogoDeHechos" as CatH
participant "hecho[i]: Hecho" as H
participant "hecho: Hecho" as He

== Registrar hecho ==

Administrador -> CH : registrar_hecho(nombre)
activate CH

CH -> CatH : buscar_hecho(nombre)
activate CatH

loop para cada hecho en CatalogoDeHechos
    
    CatH -> H : getNombre()
    activate H
    H --> CatH : <<nombre>>
        note right of CatH
            Si nombre = nombre_ingresado
            hecho = hecho[i]
            cortar iteración
        end note
    deactivate H

end

CatH --> CH : <<hecho:Hecho>>
deactivate CatH

alt Hecho no encontrado (nombre disponible)

    CH -> He : registrarHecho()
    activate He
    He --> CH : <<confirmación registro (boolean)>>
    deactivate He

    CH --> Administrador : <<registro exitoso>>

else Hecho duplicado / error

    CH --> Administrador : <<mensaje de error>>

end

deactivate CH
@enduml
