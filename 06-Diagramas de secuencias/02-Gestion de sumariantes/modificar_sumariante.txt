@startuml
actor Administrador

Administrador -> PantallaGestionSumariante : seleccionarSumariante()
Administrador -> PantallaGestionSumariante : clickModificarSumariante()

PantallaGestionSumariante -> ControladorSumariante : obtenerDatosSumariante(id)
ControladorSumariante -> Sumariante : buscarPorId(id)
Sumariante --> ControladorSumariante : «datos sumariante»
ControladorSumariante --> PantallaGestionSumariante : mostrarDatos(datos)

Administrador -> PantallaGestionSumariante : ingresarModificaciones(datosModificados)

PantallaGestionSumariante -> ControladorSumariante : modificarSumariante(datosModificados)

ControladorSumariante -> ControladorSumariante : validarDatos(datosModificados)

alt [Datos no válidos]
    ControladorSumariante --> PantallaGestionSumariante : «Error: Datos inválidos»
    PantallaGestionSumariante --> Administrador : «Corregir datos»
else [Datos válidos]
    ControladorSumariante -> Sumariante : verificarNombreExistente(datosModificados.nombre)

    alt [Nombre repetido]
        Sumariante --> ControladorSumariante : existe = true
        ControladorSumariante --> PantallaGestionSumariante : «Nombre existente»
        PantallaGestionSumariante --> Administrador : «string: Nombre repetido»
    else [Nombre libre]
        Sumariante --> ControladorSumariante : existe = false
        ControladorSumariante -> Sumariante : actualizar(datosModificados)
        Sumariante --> ControladorSumariante : «sumariante.id»
        ControladorSumariante --> PantallaGestionSumariante : mostrarConfirmación("Datos actualizados")
        PantallaGestionSumariante --> Administrador : «Sumariante modificado»
    end
end

== Flujo alternativo: cancelar ==

Administrador -> PantallaGestionSumariante : clickCancelar()
PantallaGestionSumariante --> Administrador : «Volver al menú»

@enduml
