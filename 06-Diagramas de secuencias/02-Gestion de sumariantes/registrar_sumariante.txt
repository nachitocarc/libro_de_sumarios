@startuml
actor Administrador

Administrador -> PantallaGestionSumariante : clickAgregarSumariante()

PantallaGestionSumariante -> Administrador : solicitarDatos(numero, nombre, apellido)

Administrador -> PantallaGestionSumariante : ingresarDatos(datos)

PantallaGestionSumariante -> ControladorSumariante : registrarSumariante(datos)

ControladorSumariante -> ControladorSumariante : validarDatos(datos)
alt [Campos inválidos o incompletos]
    ControladorSumariante --> PantallaGestionSumariante : «Error: Datos incompletos o inválidos»
    PantallaGestionSumariante --> Administrador : «Corregir datos»
else [Campos válidos]
    ControladorSumariante -> Sumariante : verificarNombreExistente(datos.nombre)

    alt [Nombre existente]
        Sumariante --> ControladorSumariante : existe = true
        ControladorSumariante --> PantallaGestionSumariante : «Nombre existente»
        PantallaGestionSumariante --> Administrador : «string: Nombre repetido»
    else [Nombre no existente]
        Sumariante --> ControladorSumariante : existe = false
        ControladorSumariante -> Sumariante : registrar(datos)
        Sumariante --> ControladorSumariante : «sumariante.id»
        ControladorSumariante --> PantallaGestionSumariante : mostrarConfirmación("Sumariante registrado")
        PantallaGestionSumariante --> Administrador : «Sumariante registrado»
    end
end

== Flujo alternativo: cancelar ==

Administrador -> PantallaGestionSumariante : clickCancelar()
PantallaGestionSumariante --> Administrador : «Volver al menú»

@enduml
