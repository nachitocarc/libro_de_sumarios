@startuml
actor Usuario

Usuario -> PantallaGestionDenuncias : clickNuevaDenuncia()

PantallaGestionDenuncias -> Usuario : mostrarFormulario()

Usuario -> PantallaGestionDenuncias : completarFormulario(datos)

PantallaGestionDenuncias -> ControladorDenuncia : validarDatos(datos)

alt [Datos incompletos]
    ControladorDenuncia --> PantallaGestionDenuncias : «Error: Campos obligatorios incompletos»
    PantallaGestionDenuncias --> Usuario : «Completar datos nuevamente»
else [Datos válidos]
    ControladorDenuncia -> Usuario : solicitarConfirmacion()
    
    alt [Confirmación rechazada]
        Usuario --> ControladorDenuncia : confirmar = false
        ControladorDenuncia --> PantallaGestionDenuncias : «Registro cancelado»
        PantallaGestionDenuncias --> Usuario : «Denuncia no registrada»
    else [Confirmación aceptada]
        Usuario --> ControladorDenuncia : confirmar = true

        ControladorDenuncia -> Denuncia : registrar(datos)
        Denuncia --> ControladorDenuncia : «idDenuncia»

        ControladorDenuncia --> PantallaGestionDenuncias : mostrarConfirmación("Denuncia registrada")
        PantallaGestionDenuncias --> Usuario : «Denuncia registrada»
    end
end


== Flujo alternativo: cancelar ==

Usuario -> PantallaGestionDenuncias : clickCancelar()
PantallaGestionDenuncias --> Usua
